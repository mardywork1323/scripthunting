Reverse a string

Return the characters of a string in reverse order.

Reverse a string in C · open page

reverse.c
#include <string.h>

void reverse(char *s) {
    size_t n = strlen(s);
    for (size_t i = 0; i < n / 2; i++) {
        char t = s[i];
        s[i] = s[n - 1 - i];
        s[n - 1 - i] = t;
    }
}

Check if a string is a palindrome

Ignores case, spaces and punctuation, so “A man, a plan, a canal: Panama” counts.

Check if a string is a palindrome in C · open page

palindrome.c
#include <ctype.h>
#include <stdbool.h>
#include <string.h>

bool is_palindrome(const char *s) {
    size_t i = 0, j = strlen(s);
    if (j == 0) return true;
    j--;
    while (i < j) {
        if (!isalnum((unsigned char)s[i])) { i++; continue; }
        if (!isalnum((unsigned char)s[j])) { j--; continue; }
        if (tolower((unsigned char)s[i]) != tolower((unsigned char)s[j]))
            return false;
        i++;
        j--;
    }
    return true;
}

Count how often each word appears

Split text into words and tally each one, most common first where the language makes that easy.

Count how often each word appears in C · open page

words.c
#include <stdio.h>
#include <string.h>

#define MAX_WORDS 1000

int main(void) {
    char text[] = "the cat and the hat and the bat";
    char words[MAX_WORDS][32];
    int counts[MAX_WORDS] = {0};
    int n = 0;

    for (char *w = strtok(text, " \t\n"); w; w = strtok(NULL, " \t\n")) {
        int i;
        for (i = 0; i < n; i++)
            if (strcmp(words[i], w) == 0) break;
        if (i == n) {
            if (n == MAX_WORDS) break;
            strncpy(words[n], w, 31);
            words[n][31] = '\0';
            n++;
        }
        counts[i]++;
    }
    for (int i = 0; i < n; i++)
        printf("%s: %d\n", words[i], counts[i]);
    return 0;
}

bool parse_int(const char *s, long *out) {
    char *end;
    errno = 0;
    long v = strtol(s, &end, 10);
    if (errno != 0 || end == s || *end != '\0') return false;
    *out = v;
    return true;
}

Find matches with a regular expression

Pull every number out of a sentence and check whether a string looks like an email address.

Find matches with a regular expression in C · open page

regex.c
#include <regex.h>
#include <stdio.h>

int main(void) {
    regex_t re;
    if (regcomp(&re, "[0-9]+", REG_EXTENDED) != 0) return 1;

    const char *s = "order 42 shipped in 3 days";
    regmatch_t m;
    while (regexec(&re, s, 1, &m, 0) == 0) {
        printf("%.*s\n", (int)(m.rm_eo - m.rm_so), s + m.rm_so);
        s += m.rm_eo;
    }
    regfree(&re);
    return 0;
}

Sort an array or list

Ascending, descending and by a field. Watch out for JavaScript's default string sort.

Sort an array or list in C · open page

sort.c
#include <stdio.h>
#include <stdlib.h>

static int cmp_int(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);
}

int main(void) {
    int nums[] = {5, 2, 9, 1, 7};
    size_t n = sizeof nums / sizeof nums[0];
    qsort(nums, n, sizeof nums[0], cmp_int);
    for (size_t i = 0; i < n; i++) printf("%d ", nums[i]);   /* 1 2 5 7 9 */
    return 0;
}

Remove duplicate values

Keep one copy of each value, preserving the original order where the language allows.

Remove duplicate values in C · open page

unique.c
#include <stdlib.h>

static int cmp_int(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);
}

/* Sorts arr and returns the new length with duplicates removed. */
size_t unique(int *arr, size_t n) {
    if (n == 0) return 0;
    qsort(arr, n, sizeof *arr, cmp_int);
    size_t w = 1;
    for (size_t r = 1; r < n; r++)
        if (arr[r] != arr[w - 1]) arr[w++] = arr[r];
    return w;
}

/* Requires n > 0. */
void min_max(const int *a, size_t n, int *min, int *max) {
    *min = *max = a[0];
    for (size_t i = 1; i < n; i++) {
        if (a[i] < *min) *min = a[i];
        if (a[i] > *max) *max = a[i];
    }
}

Binary search a sorted array

Find an item's index in O(log n). The array must already be sorted.

Binary search a sorted array in C · open page

bsearch.c
#include <stddef.h>

/* Returns the index of target in sorted arr, or -1. */
long binary_search(const int *arr, size_t n, int target) {
    size_t lo = 0, hi = n;
    while (lo < hi) {
        size_t mid = lo + (hi - lo) / 2;
        if (arr[mid] == target) return (long)mid;
        if (arr[mid] < target) lo = mid + 1;
        else hi = mid;
    }
    return -1;
}

function binarySearch(arr, target) {
  let lo = 0, hi = arr.length - 1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    if (arr[mid] === target) return mid;
    if (arr[mid] < target) lo = mid + 1;
    else hi = mid - 1;
  }
  return -1;
}

Quicksort

The classic divide-and-conquer sort, written out by hand for interviews and coursework.

Quicksort in C · open page

quicksort.c
static void swap(int *a, int *b) { int t = *a; *a = *b; *b = t; }

void quicksort(int *a, int lo, int hi) {
    if (lo >= hi) return;
    int pivot = a[hi], i = lo;
    for (int j = lo; j < hi; j++)
        if (a[j] < pivot) swap(&a[i++], &a[j]);
    swap(&a[i], &a[hi]);
    quicksort(a, lo, i - 1);
    quicksort(a, i + 1, hi);
}

/* quicksort(arr, 0, n - 1); */

static void QuickSort(int[] a, int lo, int hi)
{
    if (lo >= hi) return;
    int pivot = a[hi], i = lo;
    for (int j = lo; j < hi; j++)
    {
        if (a[j] < pivot)
        {
            (a[i], a[j]) = (a[j], a[i]);
            i++;
        }
    }
    (a[i], a[hi]) = (a[hi], a[i]);
    QuickSort(a, lo, i - 1);
    QuickSort(a, i + 1, hi);
}

// QuickSort(arr, 0, arr.Length - 1);

function quicksort(arr) {
  if (arr.length <= 1) return arr;
  const [pivot, ...rest] = arr;
  return [
    ...quicksort(rest.filter((x) => x < pivot)),
    pivot,
    ...quicksort(rest.filter((x) => x >= pivot)),
  ];
}

Fibonacci numbers

Iterative nth Fibonacci number. Fixed-width integers overflow after fib(93).

Fibonacci numbers in C · open page

fib.c
unsigned long long fib(unsigned n) {
    unsigned long long a = 0, b = 1;
    while (n--) {
        unsigned long long t = a + b;
        a = b;
        b = t;
    }
    return a;   /* exact up to fib(93) */
}

def is_prime(n: int) -> bool:
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    i = 3
    while i * i <= n:
        if n % i == 0:
            return False
        i += 2
    return True

long gcd(long a, long b) {
    while (b != 0) {
        long t = a % b;
        a = b;
        b = t;
    }
    return a < 0 ? -a : a;
}

Stack and queue

Last-in-first-out and first-in-first-out containers.

Stack and queue in C · open page

stack.c
#include <stdbool.h>

#define CAP 100

typedef struct {
    int items[CAP];
    int top;
} Stack;

bool push(Stack *s, int v) {
    if (s->top == CAP) return false;
    s->items[s->top++] = v;
    return true;
}

bool pop(Stack *s, int *out) {
    if (s->top == 0) return false;
    *out = s->items[--s->top];
    return true;
}

Run work in parallel

Start several jobs at once and collect their results.

Run work in parallel in C · open page

parallel.c
#include <pthread.h>
#include <stdio.h>

static void *work(void *arg) {
    int id = *(int *)arg;
    printf("thread %d running\n", id);
    return NULL;
}

int main(void) {
    pthread_t threads[4];
    int ids[4];
    for (int i = 0; i < 4; i++) {
        ids[i] = i;
        pthread_create(&threads[i], NULL, work, &ids[i]);
    }
    for (int i = 0; i < 4; i++)
        pthread_join(threads[i], NULL);
    return 0;
}
