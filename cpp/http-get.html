Make an HTTP GET request in C++

Download a page or call a JSON API.

get.cget.cppProgram.csget.mjsget.py
#include <curl/curl.h>
#include <iostream>
#include <string>

static size_t collect(char* data, size_t size, size_t n, void* out) {
    static_cast<std::string*>(out)->append(data, size * n);
    return size * n;
}

int main() {
    CURL* curl = curl_easy_init();
    if (!curl) return 1;

    std::string body;
    curl_easy_setopt(curl, CURLOPT_URL, "https://example.com");
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, collect);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, &body);
    CURLcode res = curl_easy_perform(curl);
    curl_easy_cleanup(curl);

    if (res != CURLE_OK) return 1;
    std::cout << body.size() << " bytes\n";
}
Build & run
g++ -std=c++17 get.cpp -o get -lcurl && ./get
  • Needs libcurl, e.g. apt install libcurl4-openssl-dev.
