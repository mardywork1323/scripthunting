Run work in parallel in C++

Start several jobs at once and collect their results.

parallel.cparallel.cppProgram.csparallel.mjsparallel.py
#include <future>
#include <iostream>
#include <vector>

int square(int x) { return x * x; }

int main() {
    std::vector<std::future<int>> jobs;
    for (int i = 1; i <= 4; ++i)
        jobs.push_back(std::async(std::launch::async, square, i));

    for (auto& j : jobs) std::cout << j.get() << '\n';   // 1 4 9 16
}
Build & run
g++ -std=c++17 parallel.cpp -o parallel -pthread && ./parallel
