Read user input from the console in C++

Prompt for a name and a number, then use them.

input.cinput.cppProgram.csinput.mjsinput.py
#include <iostream>
#include <string>

int main() {
    std::string name;
    std::cout << "Your name: ";
    std::getline(std::cin, name);

    int age;
    std::cout << "Your age: ";
    if (std::cin >> age)
        std::cout << "Hello, " << name << " (" << age << ")\n";
}
Build & run
g++ -std=c++17 input.cpp -o input && ./input
